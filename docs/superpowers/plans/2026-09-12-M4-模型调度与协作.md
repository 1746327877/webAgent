> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让单卡 12GB 显存下的**多模型按需调度**成为现实（ModelManager 单槽位状态机 + 显存值守 + `model_switching` 事件），并落地 **`@` 接力协作**（主智能体答完→被@智能体接力，串行换模型、消息按智能体归属）与**多模态图片理解**（VL 模型按需加载）；同时清偿 M3 首批遗留。

**Architecture:** 新增 `app/ai/model_manager.py`（单活跃槽位、asyncio 锁、model_events 落库、/api/ps 值守），runtime 在生成前 `acquire` 并对前端发 `model_switching`；`@` 接力由消息端点在主回合后串行调用 `run_generation(agent_override=..., relay_instruction=...)` 实现（单显存天然串行）；图片经 `attachments` 上传后在 VL 模型回合以 Ollama `images` 字段传入。

**Tech Stack:** M0-M3 全部沿用；后端零新依赖；前端零新依赖。

**Spec（设计文档）:** `docs/设计/07-模型动态加载与切换.md`、`06-会话与消息流.md`（接力/多模态）、`09-前端交互设计.md`（@ 面板/消息归属）；M3 终审遗留清单（台账与终审报告）。

## Global Constraints

- 沿用 M0-M3 全部约束：API 前缀 `/api/v1`；Windows + PowerShell；中文 UI；提交前缀 `feat:/fix:/chore:` 且每任务提交一次；测试输出 pristine；`ensure_ascii=False`。
- **分支策略**：`feat/m4-scheduler` 从 `feat/m3-rag` 尖端（`b700e15`）创建（堆叠；合并顺序 M0→M1→M2→M3→M4）。
- **单槽位约束**：任一时刻至多一个 LLM 处于 LOADING/LOADED；`ModelManager.acquire(model)` 在切换前 `unload` 当前；流式生成期间**不持切换锁**；切换只发生在生成开始前（回合边界）。
- `model_switching` SSE 事件载荷：`{from, to, stage}`，`stage ∈ start|done`（start=即将卸载/加载，done=目标已就绪）；事件在 `message_start` 之后、首个 token 之前。
- `@` 接力：单条消息最多 @ 2 个智能体；接力消息 `agent_id`=被@智能体、`agent_version`=其 `current_version`；接力上下文中注入指令后缀；不可用（非本人/已归档）→ `error` 事件「无法接力：智能体不可用」且跳过。
- 多模态：图片 `png/jpg/jpeg/webp` ≤5MB；有图回合使用 `settings.vision_model`（默认 `qwen2.5vl:7b`）经 ModelManager 切换；Ollama 消息 `images: [base64]` 传递。
- 显示分口径（M3 carry）：`citation.similarity` = 余弦相似度（0~1，用于 UI 展示）；`score` 仍为 RRF 排序分（保留字段供 M5）；`score_threshold` 暂不参与过滤（M5 接阈值时以 similarity 为准）。
- 新增后端依赖：无；前端依赖：无。
- 所有命令在对应目录执行（后端 `D:\code\webAgent\backend`、前端 `D:\code\webAgent\frontend`、git 在仓库根）。

```sql
        WITH semantic AS (
            SELECT id, 1 - (embedding <=> CAST(:qvec AS vector)) AS sim,
                   ROW_NUMBER() OVER (ORDER BY embedding <=> CAST(:qvec AS vector)) AS r
            FROM chunks WHERE kb_id = ANY(:kb_ids)
            ORDER BY embedding <=> CAST(:qvec AS vector) LIMIT :chan

        fused AS (
            SELECT id, SUM(1.0 / (:rrf + r)) AS rrf_score, count(*) AS channel_hits,
                   MAX(sim) AS similarity
            FROM (
                SELECT id, r, sim FROM semantic
                UNION ALL
                SELECT id, r, NULL AS sim FROM fulltext
            ) t
            GROUP BY id
        )
        -- 最终 SELECT 追加 f.similarity（现为 f.rrf_score, f.channel_hits 处）
```

```python
class RecordingProvider(FakeProvider):
    """记录调用序列；可注入加载失败。"""

    def __init__(self, script=None, fail_on: set[str] | None = None):
        super().__init__(script)
        self.calls: list[tuple] = []
        self.fail_on = fail_on or set()

    async def ensure_loaded(self, model: str) -> float:
        self.calls.append(("ensure", model))
        if model in self.fail_on:
            raise RuntimeError("oom")
        return 10.0

    async def unload(self, model: str) -> None:
        self.calls.append(("unload", model))
```

```python
import asyncio
import time
from datetime import datetime, timezone

from sqlalchemy.ext.asyncio import async_sessionmaker, AsyncSession

from app.ai.providers.base import ModelProvider
from app.core.config import settings
from app.models import ModelEvent


class ModelManager:
    """单活跃槽位：同一时刻至多一个 LLM 处于 LOADING/LOADED。"""

    def __init__(
        self,
        provider: ModelProvider,
        *,
        session_factory: async_sessionmaker[AsyncSession] | None = None,
    ):
        self.provider = provider
        self._factory = session_factory
        self._lock = asyncio.Lock()
        self._current: str | None = None
        self.last_snapshot: dict = {"loaded": [], "vram_mb": 0, "updated_at": None}

    @property
    def current(self) -> str | None:
        return self._current

    async def _log(self, model: str, action: str, trigger: str | None, **extra) -> None:
        from app.core.db import SessionLocal

        factory = self._factory or SessionLocal
        async with factory() as db:
            db.add(ModelEvent(model=model, action=action, trigger=trigger, **extra))
            await db.commit()

    async def _vram_mb(self) -> int | None:
        try:
            loaded = await self.provider.list_loaded()
            return int(sum(m.size_vram_mb for m in loaded))
        except Exception:  # noqa: BLE001
            return None

    async def acquire(self, model: str, *, trigger: str = "auto_switch") -> dict:
        async with self._lock:
            if self._current == model:
                return {"switched": False, "from": model, "to": model, "duration_ms": 0}
            prev = self._current
            vram_before = await self._vram_mb()
            if prev is not None:
                await self.provider.unload(prev)
                await self._log(prev, "unload", trigger, vram_before=vram_before)
                self._current = None
            t0 = time.monotonic()
            try:
                await self.provider.ensure_loaded(model)
            except Exception as exc:  # noqa: BLE001
                await self._log(model, "load_failed", trigger, error=str(exc)[:300])
                raise
            duration = int((time.monotonic() - t0) * 1000)
            self._current = model
            await self._log(
                model,
                "load",
                trigger,
                duration_ms=duration,
                vram_before=vram_before,
                vram_after=await self._vram_mb(),
            )
            return {"switched": True, "from": prev, "to": model, "duration_ms": duration}

    async def watch(self) -> None:
        while True:
            try:
                loaded = await self.provider.list_loaded()
                self.last_snapshot = {
                    "loaded": [
                        {"name": m.name, "size_vram_mb": m.size_vram_mb, "expires_at": m.expires_at}
                        for m in loaded
                    ],
                    "vram_mb": int(sum(m.size_vram_mb for m in loaded)),
                    "updated_at": datetime.now(timezone.utc).isoformat(),
                }
            except Exception:  # noqa: BLE001
                pass
            await asyncio.sleep(settings.vram_poll_seconds)

    async def status(self) -> dict:
        loaded = await self.provider.list_loaded()
        return {
            "current": self._current,
            "loaded": [
                {"name": m.name, "size_vram_mb": m.size_vram_mb, "expires_at": m.expires_at}
                for m in loaded
            ],
            "last_snapshot": self.last_snapshot,
        }
```
